function doPost(e) { try { var data; if (e.postData.type === 'application/json') { data = JSON.parse(e.postData.contents); } else { // Form data data = e.parameter; } var actNumber = getNextActNumber(data.date); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_JOURNAL); sheet.appendRow([data.date, data.company, data.type, data.name, data.serial, data.note, data.employee, actNumber]); var pdf = generateActPdf(data, actNumber); var attachments = [pdf]; if (data.photoType && data.photoType.length > 10) attachments.push(Utilities.newBlob(Utilities.base64Decode(data.photoType), 'image/jpeg', 'Фото_тип_' + data.serial + '.jpg')); if (data.photoSerial && data.photoSerial.length > 10) attachments.push(Utilities.newBlob(Utilities.base64Decode(data.photoSerial), 'image/jpeg', 'Фото_номер_' + data.serial + '.jpg')); GmailApp.sendEmail(EMAIL_TO, 'Акт приёмки СИ ' + actNumber + ': ' + data.name + ' №' + data.serial, 'Сформирован акт приёмки.\n\nНомер акта: ' + actNumber + '\nДата: ' + data.date + '\nКомпания: ' + data.company + '\nТип: ' + data.type + '\nНаименование: ' + data.name + '\nЗаводской номер: ' + data.serial + '\nПримечание: ' + data.note + '\nПринял: ' + data.employee, { attachments: attachments }); return ContentService.createTextOutput('OK').setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput('ERROR: ' + error.toString()).setMimeType(ContentService.MimeType.TEXT); } }
